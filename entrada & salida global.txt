Sub registrasalida(ByVal instruccion As String, ByVal instruccion2 As String, ByVal instruccion3 As String, ByVal instruccion4 As String)
        cierraconexion()
        abreconexion()
        sqltr = Nothing
        Dim comando As PsqlCommand = cadenaconexion.CreateCommand
        sqltr = cadenaconexion.BeginTransaction
        comando.Connection = cadenaconexion
        comando.Transaction = sqltr
        If instruccion = "" And instruccion2 = "" Then
            comando.CommandText = instruccion3
            comando.ExecuteNonQuery()
            comando.CommandText = instruccion4
            comando.ExecuteNonQuery()
        Else
            comando.CommandText = instruccion
            comando.ExecuteNonQuery()
            comando.CommandText = instruccion2
            comando.ExecuteNonQuery()
            comando.CommandText = instruccion3
            comando.ExecuteNonQuery()
            comando.CommandText = instruccion4
            comando.ExecuteNonQuery()
        End If
        sqltr.Commit()

    End Sub
